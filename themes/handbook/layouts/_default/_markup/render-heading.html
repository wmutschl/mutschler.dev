{{- /* Headings with Willi's own section numbers typeset as structure: "3.1 General" and "Step 2 (optional): Install …" become a number span (hung in the gutter on wide screens) and a title span. "Step" stays in the accessible name; "(optional)" moves to the end in italic. Headings without a number are left as written. The id is unchanged, so existing links keep working. */ -}} {{- $text := string .Text -}} {{- $no := "" -}} {{- $step := false -}} {{- $opt := false -}} {{- $rest := $text -}} {{- with findRESubmatch `^(?i:step)\s+(\d+[a-z]?)\s*(\((?i:optional)\))?\s*:\s*(.+)$` $text 1 -}} {{- $m := index . 0 -}} {{- $no = index $m 1 -}} {{- $step = true -}} {{- $opt = ne (index $m 2) "" -}} {{- $rest = index $m 3 -}} {{- else -}} {{- with findRESubmatch `^(\d+(?:\.\d+)*)\s+(.+)$` $text 1 -}} {{- $m := index . 0 -}} {{- $no = index $m 1 -}} {{- $rest = index $m 2 -}} {{- end -}} {{- end -}} {{- with $no }}{{ if $step }}Step {{ end }}{{ . }}{{ if $step }}:{{ end }} {{ end -}} {{ $rest | safeHTML }}{{ if $opt }} (optional){{ end -}} {{- /* trailing */ -}}
