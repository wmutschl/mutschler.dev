{{- /* The page TOC (h2–h3) as nested lists, with Willi's section numbers ("3.1 …", "Step 2: …") moved into their own span so they can align in a column. A space separates number and title in the text, so screen readers and copied text do not run them together (the flex layout does not render it). "Step" stays for screen readers; "(optional)" moves to the end of the entry. No ids, so it can appear more than once on a page. Usage: {{ partial "handbook/toc.html" . }} */ -}} {{- $toc := partial "base/toc.html" . | string -}} {{- $toc = replaceRE `(?i:step)\s+(\d+[a-z]?)\s*\((?i:optional)\)\s*:\s*(.*?)` `Step ${2} ${3} (optional)` $toc -}} {{- $toc = replaceRE `(?i:step)\s+(\d+[a-z]?)\s*:\s*(.*?)` `Step ${2} ${3}` $toc -}} {{- $toc = replaceRE `(\d+(?:\.\d+)*)\s+(.*?)` `${2} ${3}` $toc -}} {{- $toc = replaceRE `([^<].*?)` `${2}` $toc -}} {{- $toc | safeHTML -}}
